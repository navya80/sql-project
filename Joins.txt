create database jfs3435joins; 

use jfs3435joins;


CREATE TABLE student (
    student_id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(50) not null
);


CREATE TABLE mess (
    mess_id INT PRIMARY KEY AUTO_INCREMENT,
    mess_type ENUM('Hostel','Day scholar') NOT NULL,
    student_id INT UNIQUE,
    FOREIGN KEY (student_id) REFERENCES student(student_id)
    ON DELETE CASCADE
    ON UPDATE CASCADE
);


CREATE TABLE sports (
    sports_id INT PRIMARY KEY AUTO_INCREMENT,
    sports_type SET('Cricket','Tennis') NOT NULL,
    student_id INT UNIQUE,
    FOREIGN KEY (student_id) REFERENCES student(student_id)
    ON DELETE CASCADE
    ON UPDATE CASCADE
);

Inner join:
select * from mess join sports on mess.student_id =sports.student_id;
select * from mess m join sports s on m.student_id =s.student_id;
select * from mess as m join sports as s on m.student_id =s.student_id;
Equi join:
select * from mess m, sports s where m.student_id = s.student_id;


Natural join: select * from mess natural join sports;


Left join:
select * from mess Left join sports on mess.student_id =sports.student_id;
select * from mess as m left join sports as s on m.student_id =s.student_id;


Right join:
select * from mess right join sports on mess.student_id = sports.student_id;



Full Join:
select * from mess left join sports on mess.student_id = sports.student_id where sports.student_id is null;


Left Exclusive Join:
select * from mess left join sports on mess.student_id = sports.student_id where sports.student_id is null;

Right Exclusive Join:
select * from mess right join sports  on mess.student_id =sports.student_id where  mess.student_id is null;


Full Exclusive Join:
